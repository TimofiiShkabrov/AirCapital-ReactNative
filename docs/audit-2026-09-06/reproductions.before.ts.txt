// Audit probes: these assertions document EXISTING defects, not desired behavior.
// All credentials, balances, storage and exchange responses are synthetic.
import { afterEach, beforeEach, describe, expect, it, vi } from 'vitest';
import { saveAccount, deleteAccount, getAllAccounts, loadKeys } from '../../src/services/secureStore';
import { addSnapshot, getSnapshots } from '../../src/services/balanceHistory';
import { usePortfolioStore } from '../../src/store/portfolioStore';
import * as Bybit from '../../src/api/bybit';

const memory = vi.hoisted(() => ({
  metadata: new Map<string, string>(),
  secrets: new Map<string, string>(),
  rejectDelete: false,
}));

vi.mock('@react-native-async-storage/async-storage', () => ({
  default: {
    getItem: vi.fn((key: string) => Promise.resolve(memory.metadata.get(key) ?? null)),
    setItem: vi.fn(async (key: string, value: string) => { memory.metadata.set(key, value); }),
    removeItem: vi.fn(async (key: string) => { memory.metadata.delete(key); }),
  },
}));
vi.mock('expo-secure-store', () => ({
  getItemAsync: vi.fn((key: string) => Promise.resolve(memory.secrets.get(key) ?? null)),
  setItemAsync: vi.fn(async (key: string, value: string) => { memory.secrets.set(key, value); }),
  deleteItemAsync: vi.fn(async (key: string) => {
    if (memory.rejectDelete) throw new Error('Synthetic secure storage failure');
    memory.secrets.delete(key);
  }),
}));
vi.mock('../../src/i18n', () => ({ default: { t: (key: string) => key } }));

const keys = { apiKey: 'audit-placeholder', secretKey: 'audit-placeholder' };

beforeEach(() => {
  memory.metadata.clear();
  memory.secrets.clear();
  memory.rejectDelete = false;
  usePortfolioStore.setState(usePortfolioStore.getInitialState(), true);
  vi.useFakeTimers();
  vi.setSystemTime(new Date('2026-09-06T10:00:00Z'));
  // Any accidentally unmocked network operation fails locally.
  vi.stubGlobal('fetch', vi.fn(async () => { throw new Error('Network prohibited in audit'); }));
});
afterEach(() => {
  vi.restoreAllMocks();
  vi.unstubAllGlobals();
  vi.useRealTimers();
});

function healthyBybit() {
  const wallet = vi.spyOn(Bybit, 'fetchWallet').mockResolvedValue({
    data: { retCode: 0, result: { list: [{ accountType: 'UNIFIED', totalEquity: '100', coin: [] }] } },
  });
  vi.spyOn(Bybit, 'fetchPositions').mockResolvedValue({ data: { retCode: 0, result: { list: [] } } });
  vi.spyOn(Bybit, 'fetchAllCoinsBalance').mockResolvedValue({ data: { retCode: 0, result: { balance: [] } } });
  vi.spyOn(Bybit, 'fetchEarnPositions').mockImplementation(async (_keys, category) => ({
    data: { retCode: 0, result: { list: category === 'FlexibleSaving' ? [{ coin: 'USDT', amount: '10' }] : [] } },
  }));
  return wallet;
}

describe('Reproduction of defects observed on 2026-09-06', () => {
  it('S-02: deletion reports success while credentials remain when SecureStore fails', async () => {
    const account = await saveAccount(keys, 'bybit');
    memory.rejectDelete = true;
    await expect(deleteAccount(account)).resolves.toBeUndefined();
    expect(await getAllAccounts()).toEqual([]);
    expect(await loadKeys(account)).toEqual({ ...keys, passphrase: undefined });
  });

  it('D-05: concurrent account saves lose one metadata record and leave its credentials', async () => {
    await Promise.all([saveAccount(keys, 'binance'), saveAccount(keys, 'okx')]);
    expect(await getAllAccounts()).toHaveLength(1);
    expect(memory.secrets.size).toBe(4);
  });

  it('D-03: history hides a genuine zero balance even after the suppression window', async () => {
    await addSnapshot({ type: 'total' }, 100);
    vi.setSystemTime(new Date('2026-09-06T11:00:00Z'));
    await addSnapshot({ type: 'total' }, 0);
    const raw = JSON.parse(memory.metadata.get('aircapital.balanceSnapshots.v1')!);
    expect(raw).toHaveLength(2);
    expect((await getSnapshots({ type: 'total' })).map(s => s.balanceUSDT)).toEqual([100]);
  });

  it('D-01: refreshing an unchanged Bybit Earn position increases the reported capital', async () => {
    const account = await saveAccount(keys, 'bybit');
    healthyBybit();
    await usePortfolioStore.getState().loadData();
    expect(usePortfolioStore.getState().getAccountBalance(account)).toBe(110);
    await usePortfolioStore.getState().loadData();
    expect(usePortfolioStore.getState().getAccountBalance(account)).toBe(120);
    expect(usePortfolioStore.getState().bybitData[account.id].earnPositions).toHaveLength(2);
  });

  it('D-02: failed synchronization writes the old balance with a new timestamp', async () => {
    await saveAccount(keys, 'bybit');
    const wallet = healthyBybit();
    await usePortfolioStore.getState().loadData();
    vi.setSystemTime(new Date('2026-09-06T11:00:00Z'));
    wallet.mockResolvedValue({ error: { code: 'tooManyRequests' } });
    await usePortfolioStore.getState().loadData();
    const snapshots = await getSnapshots({ type: 'total' });
    expect(snapshots.at(-1)?.balanceUSDT).toBe(110);
    expect(snapshots.at(-1)?.timestamp).toBe('2026-09-06T11:00:00.000Z');
    expect(Object.keys(usePortfolioStore.getState().accountFailures)).toHaveLength(1);
  });

  it('D-04: an exchange envelope error can reject refresh and leave loading active', async () => {
    await saveAccount(keys, 'bybit');
    healthyBybit().mockResolvedValue({ data: { retCode: 10003, result: {} } } as never);
    await expect(usePortfolioStore.getState().loadData()).rejects.toThrow();
    expect(usePortfolioStore.getState().isLoading).toBe(true);
  });

  it('D-06: a funding wallet with only BTC contributes zero to the OKX total', async () => {
    const account = await saveAccount(keys, 'okx');
    usePortfolioStore.setState({ accounts: [account], okxData: { [account.id]: {
      fundingBalances: [{ ccy: 'BTC', bal: '1', availBal: '1', frozenBal: '0' }],
    } } });
    expect(usePortfolioStore.getState().getAccountBalance(account)).toBe(0);
  });
});
